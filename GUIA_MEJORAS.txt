FORTALEZAS:
  + Datos reales y actualizados (INDEC, BCRA, Min. Economía)
  + Diseño oscuro consistente con variables CSS propias

  - El globo 3D no muestra datos georreferenciados
  - No hay transiciones entre páginas
  - La calculadora de instrumentos podría tener gráfico interactivo tipo TradingView
  - Los números en KPIs no tienen animación de conteo al cargar
  - Mobile: algunos charts se ven comprimidos
  - No hay comando rápido (⌘K está en el navbar pero no implementado visualmente)

A) Reemplazar useLiveData.ts completamente
   -------------------------------------------------------
   // ANTES (src/hooks/useLiveData.ts — probablemente usa useEffect + fetch manual)

   // DESPUÉS con SWR:
   import useSWR from 'swr'

   const fetcher = (url: string) => fetch(url).then(r => r.json())

   export function useLiveKPIs() {
     const { data, error, isLoading } = useSWR('/api/kpis', fetcher, {
       refreshInterval: 60000,      // actualizar cada 60s (igual que ahora)
       revalidateOnFocus: true,      // actualizar cuando el usuario vuelve a la tab
       dedupingInterval: 30000,      // no re-fetchar si ya hay dato fresco
     })
     return { data, error, isLoading }
   }

   // Para el dólar blue (más volátil, refresh más rápido):
   export function useDolarBlue() {
     const { data } = useSWR('/api/dolar', fetcher, {
       refreshInterval: 30000,
     })
     return data
   }

B) Beneficio clave para MacroLibre:
   - Si el usuario navega a /inflacion y vuelve al home, los KPIs
     no re-fetchan — usan el caché. Experiencia instantánea.
   - Si la API de dólar falla, SWR muestra el último dato conocido

   // Nuevo componente: src/components/charts/InstrumentChart.tsx
   'use client'
   import { createChart, ColorType } from 'lightweight-charts'
   import { useEffect, useRef } from 'react'

   export function InstrumentChart({ series }: { series: ChartSeries[] }) {
     const chartRef = useRef<HTMLDivElement>(null)

     useEffect(() => {
       if (!chartRef.current) return
       const chart = createChart(chartRef.current, {
         layout: {
           background: { type: ColorType.Solid, color: 'transparent' },
           textColor: 'var(--fg-2)',
         },
         grid: {
           vertLines: { color: 'var(--line-1)' },
           horzLines: { color: 'var(--line-1)' },
         },
         width: chartRef.current.clientWidth,
         height: 320,
       })
       // agregar series...
       return () => chart.remove()
     }, [series])

     return <div ref={chartRef} />
   }

   'use client'
   import Globe from 'globe.gl'
   import { useEffect, useRef } from 'react'

   export function MacroGlobe() {
     const mountRef = useRef<HTMLDivElement>(null)

     useEffect(() => {
       if (!mountRef.current) return
       const globe = Globe()(mountRef.current)
         .globeImageUrl('//unpkg.com/three-globe/example/img/earth-dark.jpg')
         .backgroundColor('rgba(0,0,0,0)')
         .pointsData(provinciasPBI)
         .pointColor(() => 'var(--celeste)')
         .pointAltitude('pbi_share')
         .pointRadius(0.4)

       return () => globe._destructor?.()
     }, [])

     return <div ref={mountRef} style={{ width: '100%', height: '500px' }} />
   }

NOTA: globe.gl usa Three.js internamente. Dado que ya tenés
